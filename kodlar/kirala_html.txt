    <script>
        let selectedBook = null;
        const params = new URLSearchParams(window.location.search);
        const bookId = parseInt(params.get('id'));
        const summaryContainer = document.getElementById('summaryCard');
        const submitBtn = document.getElementById('submitBtn');

        document.addEventListener('DOMContentLoaded', async function() {
            await loadBookData();
        });

        async function loadBookData() {
            try {
                const response = await fetch('api/books.php');
                const books = await response.json();
                
                selectedBook = books.find(b => b.id === bookId);
                
                if (selectedBook) {
                    if (selectedBook.stok <= 0) {
                        showBookUnavailable();
                    } else {
                        renderSummary(15); 
                    }
                } else {
                    showBookNotFound();
                }
            } catch (error) {
                console.error('Kitap bilgisi yüklenirken hata:', error);
                showBookNotFound();
            }
        }

        async function submitRental(rentalData) {
            try {
                const response = await fetch('api/rent.php', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify(rentalData)
                });
                
                const result = await response.json();
                return result;
            } catch (error) {
                console.error('Kiralama işlemi hatası:', error);
                return { success: false, message: 'Sunucu hatası: ' + error.message };
            }
        }

        
        function showBookUnavailable() {
            document.querySelector('.form-card').style.display = 'none';
            summaryContainer.innerHTML = `
                <div style="text-align: center; padding: 40px 20px;">
                    <i class="fas fa-times-circle" style="font-size: 3rem; color: var(--danger); margin-bottom: 20px;"></i>
                    <h3 style="color: var(--danger);">Kitap Müsait Değil</h3>
                    <p style="color: #666; margin: 10px 0;">Bu kitap şu anda başka bir okuyucuda.</p>
                    <a href="index.html" class="confirm-btn" style="text-decoration: none; display: inline-block; padding: 10px 20px; margin-top: 20px;">
                        Başka Kitaplara Göz At
                    </a>
                </div>
            `;
        }

        function showBookNotFound() {
            document.querySelector('.form-card').style.display = 'none';
            summaryContainer.innerHTML = `
                <div style="text-align: center; padding: 40px 20px;">
                    <i class="fas fa-search" style="font-size: 3rem; color: #999; margin-bottom: 20px;"></i>
                    <h3>Kitap Bulunamadı</h3>
                    <p style="color: #666; margin: 10px 0;">Aradığınız kitap kütüphanemizde mevcut değil.</p>
                    <a href="index.html" class="confirm-btn" style="text-decoration: none; display: inline-block; padding: 10px 20px; margin-top: 20px;">
                        Kütüphaneye Dön
                    </a>
                </div>
            `;
        }

        function renderSummary(cargoPrice) {
            const total = parseFloat(cargoPrice).toFixed(2);
            
            summaryContainer.innerHTML = `
                <div class="summary-header">
                    <img src="${selectedBook.kapak_resmi}" class="book-cover-preview" alt="${selectedBook.baslik}">
                    <h3>${selectedBook.baslik}</h3>
                    <p style="color: var(--wood-light); font-style: italic;">${selectedBook.yazar}</p>
                </div>
                
                <ul class="summary-list">
                    <li class="summary-item">
                        <span>Kategori</span>
                        <span>${selectedBook.kategori}</span>
                    </li>
                    <li class="summary-item">
                        <span>Kiralama Süresi</span>
                        <span>15 Gün</span>
                    </li>
                    <li class="summary-item">
                        <span>Kitap Bedeli</span>
                        <span>Ücretsiz</span>
                    </li>
                    <li class="summary-item">
                        <span>Kargo Ücreti</span>
                        <span id="cargoDisplay">${cargoPrice > 0 ? cargoPrice + '.00 TL' : 'Ücretsiz'}</span>
                    </li>
                </ul>

                <div class="total-price">
                    <span>Toplam Tutar</span>
                    <span id="totalDisplay" style="color: var(--accent);">${total} TL</span>
                </div>
            `;
        }

        function updateCheckoutState() {
            const select = document.getElementById('cargoSelect');
            const price = parseInt(select.value);
            
            renderSummary(price);

            const infoBox = document.getElementById('deliveryInfoBox');
            const infoTitle = document.getElementById('infoTitle');
            const infoDesc = document.getElementById('infoDesc');
            const infoIcon = infoBox.querySelector('i');

            if (price === 15) {
                infoBox.style.backgroundColor = '#fff8e1';
                infoBox.style.borderLeftColor = 'var(--accent)';
                infoIcon.className = 'fas fa-box-open';
                infoIcon.style.color = 'var(--accent)';
                infoTitle.innerText = 'Standart Teslimat';
                infoDesc.innerText = 'Kitabınız sterilize edilmiş kutusunda, 3 iş günü içinde teslim edilir.';
            } else {
                infoBox.style.backgroundColor = '#e3f2fd';
                infoBox.style.borderLeftColor = 'var(--info-blue)';
                infoIcon.className = 'fas fa-rocket';
                infoIcon.style.color = 'var(--info-blue)';
                infoTitle.innerText = 'Hızlı Teslimat (Express)';
                infoDesc.innerText = 'Siparişiniz öncelikli hazırlanır ve 24 saat içinde adresinize ulaştırılır.';
            }
        }

        document.getElementById('rentalForm').addEventListener('submit', async function(e) {
            e.preventDefault();
            
            if (!selectedBook) {
                showError('Kitap bilgisi bulunamadı.');
                return;
            }

            submitBtn.disabled = true;
            submitBtn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> İşleniyor...';

            const rentalData = {
                kitap_id: selectedBook.id,
                ad: document.getElementById('firstName').value,
                soyad: document.getElementById('lastName').value,
                email: document.getElementById('email').value,
                telefon: document.getElementById('phone').value,
                adres: document.getElementById('address').value,
                kargo_ucreti: parseInt(document.getElementById('cargoSelect').value),
                kargo_tipi: document.getElementById('cargoSelect').selectedIndex === 0 ? 'standart' : 'hizli'
            };

            try {
                const result = await submitRental(rentalData);

console.log("API Yanıtı:", result);

if (result.success) {
    showSuccessModal();
} else {
    const debugInfo = result.debug ? ` (Debug: ${JSON.stringify(result.debug)})` : '';
    showError(result.message + debugInfo || 'Kiralama işlemi başarısız oldu.');
}
            } catch (error) {
                showError('Bir hata oluştu: ' + error.message);
            } finally {
                submitBtn.disabled = false;
                submitBtn.innerHTML = 'Kiralama İşlemini Tamamla <i class="fas fa-arrow-right"></i>';
            }
        });

        
        function openContractModal(e) {
            e.preventDefault();
            document.getElementById('contractModal').style.display = 'flex';
        }

        function closeContractModal() {
            document.getElementById('contractModal').style.display = 'none';
        }

        function showSuccessModal() {
            document.getElementById('successModal').style.display = 'flex';
            setTimeout(() => {
                window.location.href = "index.html";
            }, 3000);
        }

        function showError(message) {
            document.getElementById('errorMessage').textContent = message;
            document.getElementById('errorModal').style.display = 'flex';
        }

        function closeErrorModal() {
            document.getElementById('errorModal').style.display = 'none';
        }

        window.onclick = function(event) {
            const contractModal = document.getElementById('contractModal');
            const errorModal = document.getElementById('errorModal');
            
            if (event.target == contractModal) {
                contractModal.style.display = "none";
            }
            if (event.target == errorModal) {
                errorModal.style.display = "none";
            }
        }
    </script>
