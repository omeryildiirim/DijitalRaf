    async function checkLoginStatus() {
        try {
            const response = await fetch('api/check_session.php');
            const result = await response.json();
            
            console.log("Oturum kontrolü:", result);
            
            if(!result.logged_in) {
                window.location.href = 'index.html';
                return false;
            }
            return result.user;
        } catch (error) {
            console.error('Giriş kontrolü hatası:', error);
            window.location.href = 'index.html';
            return false;
        }
    }

    async function logout() {
        try {
            const response = await fetch('api/logout.php');
            const result = await response.json();
            
            if(result.success) {
                actionSuccess('Çıkış yapıldı. Ana sayfaya yönlendiriliyorsunuz...');
                setTimeout(() => {
                    window.location.href = 'index.html';
                }, 1500);
            }
        } catch (error) {
            console.error('Çıkış hatası:', error);
            window.location.href = 'index.html';
        }

    function renderRentals(rentals) {
        const container = document.getElementById('activeRentals');
        container.innerHTML = "";
        
        if(rentals.length === 0) {
            container.innerHTML = `
                <div class="empty-state">
                    <i class="fas fa-book-open"></i>
                    <h3 style="margin-bottom: 10px;">Henüz kitap kiralamamışsınız</h3>

        }

        rentals.forEach((rental) => {
            const kiralamaTarihi = new Date(rental.kiralama_tarihi);
            const simdi = new Date();
            const gunFarki = Math.floor((simdi - kiralamaTarihi) / (1000 * 60 * 60 * 24));
            const kalanGun = 14 - gunFarki;
            
            const isUrgent = kalanGun <= 3;
            const tagClass = isUrgent ? 'tag-urgent' : 'tag-normal';
            const tagText = isUrgent ? `Son ${kalanGun} Gün!` : `${kalanGun} Gün Kaldı`;

            const card = document.createElement('div');
            card.className = 'rent-card';
            card.innerHTML = `
                <img src="${rental.kapak_resmi || 'https://images.unsplash.com/photo-1544716278-ca5e3f4abd8c?w=200'}" class="rent-img" alt="${rental.baslik}" onerror="this.src='https://images.unsplash.com/photo-1544716278-ca5e3f4abd8c?w=200'">
                <div class="rent-info">
                    <span class="rent-tag ${tagClass}">${tagText}</span>
                    <div class="rent-title">${rental.baslik}</div>
                    <div class="rent-author">${rental.yazar}</div>
                    <button class="return-btn" onclick="returnBook(${rental.id})">
                        <i class="fas fa-undo"></i> İade Et
                    </button>
                </div>
            `;
            container.appendChild(card);
        });
    }

let returnProcessBooks = [];
async function returnBook(kiralamaId) {
    if(confirm("İade talebi oluşturmak üzeresiniz. Onaylıyor musunuz?")) {
        try {
            const randomSayi = Math.floor(1000 + Math.random() * 9000);
            const iadeKodu = 'RF-' + randomSayi;

            try {
                const response = await fetch('api/get_user_rentals.php');
                const result = await response.json();
                
                if(result.success && result.data) {
                    const kitap = result.data.find(r => r.id == kiralamaId);
                    if(kitap) {
                        const newReturnProcess = {
                            id: Date.now(), 
                            kiralamaId: kiralamaId,
                            baslik: kitap.baslik,
                            yazar: kitap.yazar,
                            kapak_resmi: kitap.kapak_resmi,
                            returnCode: iadeKodu,
                            returnDate: new Date().toISOString().split('T')[0],
                            status: 'pending', 
                            timeline: [
                                { step: "İade Talebi Oluşturuldu", date: new Date().toISOString().split('T')[0], active: true },
                                { step: "Kargoya Verildi", date: null, active: false },
                                { step: "Kargoda", date: null, active: false },
                                { step: "İnceleniyor", date: null, active: false },
                                { step: "Tamamlandı", date: null, active: false }
                            ]
                        };
                        
                        returnProcessBooks.push(newReturnProcess);
                        
                        try {
                            const iadeResponse = await fetch('api/return_book.php', {
                                method: 'POST',
                                headers: { 'Content-Type': 'application/json' },
                                body: JSON.stringify({ 
                                    kiralama_id: kiralamaId,
                                    return_code: iadeKodu 
                                })
                            });
                            
                            const iadeResult = await iadeResponse.json();
                            
                            if(iadeResult.success) {
                                loadUserRentals();
                                
                                document.getElementById('returnProcessSection').style.display = 'flex';
                                renderReturnProcess();
                                
                                document.getElementById('generatedReturnCode').innerText = iadeKodu;
                                
                                openModal('returnCodeModal');
                                
                                actionSuccess("İade talebiniz alındı, lütfen kodu not edin.");
                            } else {
                                alert('İade işlemi başarısız: ' + iadeResult.error);
                            }
                            
                        } catch (apiError) {
                            console.error("API hatası:", apiError);
                            alert('İade işlemi sırasında bir hata oluştu.');
                        }
                    }
                }
            } catch (error) {
                console.error("Kitap bilgisi alınamadı:", error);
            }

        } catch (error) {
            alert('İade kodu oluşturulurken hata oluştu: ' + error);
        }
    }
}

function renderReturnProcess() {
    const container = document.getElementById('returnProcessGrid');
    container.innerHTML = '';
    
    if(returnProcessBooks.length === 0) {
        container.innerHTML = `
            <div class="empty-state">
                <i class="fas fa-box-open"></i>
                <h3 style="margin-bottom: 10px;">İade sürecinde kitap yok</h3>
                <p>İade ettiğiniz kitaplar burada görünecektir.</p>
            </div>
        `;
        document.getElementById('returnProcessSection').style.display = 'none';
        return;
    }
    
    returnProcessBooks.forEach((book, index) => {
        const statusClass = `status-${book.status}`;
        const statusIcon = getReturnStatusIcon(book.status);
        const statusText = getReturnStatusText(book.status);
        
        const card = document.createElement('div');
        card.className = 'rent-card';
        card.innerHTML = `
            <img src="${book.kapak_resmi || 'https://images.unsplash.com/photo-1544716278-ca5e3f4abd8c?w=200'}" 
                 class="rent-img" alt="${book.baslik}" 
                 onerror="this.src='https://images.unsplash.com/photo-1544716278-ca5e3f4abd8c?w=200'">
            <div class="rent-info">
                <div class="rent-title">${book.baslik}</div>
                <div class="rent-author">${book.yazar}</div>
                
                <!-- İade Süreci Bilgisi -->
                <div class="return-process">
                    <div class="process-tag ${statusClass}">
                        <i class="${statusIcon}"></i> ${statusText}
                    </div>
                    <div class="process-info">
                        <i class="fas fa-barcode"></i> İade Kodu: <strong>${book.returnCode}</strong>
                        <br>
                        <i class="far fa-calendar"></i> Talep Tarihi: ${book.returnDate}
                    </div>
                </div>
                
                <!-- Zaman Çizelgesi -->
                <div class="process-timeline">
                    ${book.timeline.map((item, idx) => `
                        <div class="timeline-item">
                            <div class="timeline-dot ${item.active ? 'active' : 'pending'}"></div>
                            <div class="timeline-content">
                                ${item.step}
                                ${item.date ? `<div class="timeline-date">${item.date}</div>` : ''}
                            </div>
                        </div>
                    `).join('')}
                </div>
                
                <!-- Durum Güncelleme Butonları (Demo için) -->
                <div style="display: flex; gap: 10px; margin-top: 15px;">
                    ${book.status === 'pending' ? `
                       
                    ` : ''}
                    ${book.status === 'shipped' ? `
                        <button class="btn btn-outline" onclick="updateReturnStatus(${index}, 'inreview')" style="font-size: 0.8rem; padding: 8px 12px;">
                            <i class="fas fa-search"></i> İncelemeye Al
                        </button>
                    ` : ''}
                    ${book.status === 'inreview' ? `
                        <button class="btn btn-outline" onclick="updateReturnStatus(${index}, 'completed')" style="font-size: 0.8rem; padding: 8px 12px;">
                            <i class="fas fa-check"></i> Tamamla
                        </button>
                    ` : ''}
                </div>
            </div>
        `;
        container.appendChild(card);
    });
}

function getReturnStatusIcon(status) {
    const icons = {
        'pending': 'fas fa-clock',
        'shipped': 'fas fa-shipping-fast',
        'inreview': 'fas fa-search',
        'completed': 'fas fa-check-circle'
    };
    return icons[status] || 'fas fa-info-circle';
}

function getReturnStatusText(status) {
    const texts = {
        'pending': 'Kargoya Verilmesi Bekleniyor',
        'shipped': 'Kargoya Verildi',
        'inreview': 'İnceleniyor',
        'completed': 'Tamamlandı'
    };
    return texts[status] || 'Bilinmiyor';
}

function updateReturnStatus(bookIndex, newStatus) {
    const book = returnProcessBooks[bookIndex];
    book.status = newStatus;
    
    const statusIndex = {
        'pending': 0,
        'shipped': 1,
        'inreview': 3,
        'completed': 4
    }[newStatus];
    
    for(let i = 0; i <= statusIndex; i++) {
        if(!book.timeline[i].date && i > 0) {
            book.timeline[i].date = new Date().toISOString().split('T')[0];
        }
        book.timeline[i].active = true;
    }
    
    for(let i = statusIndex + 1; i < book.timeline.length; i++) {
        book.timeline[i].active = false;
    }
    
    if(newStatus === 'shipped') {
        book.timeline[2].active = true;
        book.timeline[2].date = new Date().toISOString().split('T')[0];
    }
    
    renderReturnProcess();
    actionSuccess(`İade durumu "${getReturnStatusText(newStatus)}" olarak güncellendi.`);
}

function clearReturnProcess() {
    if(returnProcessBooks.length === 0) {
        actionSuccess("Zaten temiz!");
        return;
    }
    
    if(confirm(`İade sürecindeki ${returnProcessBooks.length} kitabı temizlemek istiyor musunuz?\n\nBu işlem geri alınamaz!`)) {
        localStorage.removeItem('dijitalraf_return_process');

        document.getElementById('returnProcessGrid').innerHTML = '';
        
        actionSuccess(`${returnProcessBooks.length} kitap iade sürecinden temizlendi.`);
        
        console.log("İade süreci temizlendi.");
    }
}

    }
