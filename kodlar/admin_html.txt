        async function getBooksFromAPI() {
            try {
                const response = await fetch('api/books.php');
                return await response.json();
            } catch (error) {
                console.error('Kitaplar yüklenirken hata:', error);
                return [];
            }
        }

        async function getRentalsFromAPI() {
            try {
                const response = await fetch('api/admin_rentals.php');
                return await response.json();
            } catch (error) {
                console.error('Kiralamalar yüklenirken hata:', error);
                return [];
            }
        }

        async function addBookToAPI(bookData) {
            try {
                const response = await fetch('api/admin_add_book.php', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify(bookData)
                });
                return await response.json();
            } catch (error) {
                console.error('Kitap eklenirken hata:', error);
                return { success: false, message: 'Sunucu hatası' };
            }
        }

        async function updateBookInAPI(bookData) {
            try {
                const response = await fetch('api/admin_edit_book.php', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify(bookData)
                });
                return await response.json();
            } catch (error) {
                console.error('Kitap güncellenirken hata:', error);
                return { success: false, message: 'Sunucu hatası' };
            }
        }

        async function deleteBookFromAPI(bookId) {
            try {
                const response = await fetch('api/admin_delete_book.php', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify({ id: bookId })
                });
                return await response.json();
            } catch (error) {
                console.error('Kitap silinirken hata:', error);
                return { success: false, message: 'Sunucu hatası' };
            }
        }

        async function returnBookAPI(rentalId) {
            try {
                const response = await fetch('api/admin_return_book.php', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                    },
                    body: JSON.stringify({ rental_id: rentalId })
                });
                return await response.json();
            } catch (error) {
                console.error('İade işlemi hatası:', error);
                return { success: false, message: 'Sunucu hatası' };
            }
        }

      
        async function updateStats() {
            const books = await getBooksFromAPI();
            const rentals = await getRentalsFromAPI();
            const members = await getMembersFromAPI();
            
            const totalBooks = books.length;
            const activeRentals = rentals.filter(r => r.durum === 'active').length;
            const totalMembers = members.length;
            
            document.getElementById('totalBooks').textContent = totalBooks;
            document.getElementById('activeRentals').textContent = activeRentals;
            document.getElementById('totalMembers').textContent = totalMembers;
            document.getElementById('lateReturns').textContent = '0';
        }

        async function renderRentals() {
            const tbody = document.getElementById('rentalTableBody');
            tbody.innerHTML = "<tr><td colspan='5'>Yükleniyor...</td></tr>";
            
            const rentals = await getRentalsFromAPI();
            const dataToShow = isRentalsExpanded ? rentals : rentals.slice(0, 5);

            tbody.innerHTML = '';
            
            dataToShow.forEach((rental) => {
                const statusHtml = rental.durum === 'active' 
                    ? `<span style="color:var(--success);"><i class="fas fa-check-circle"></i> Aktif</span>` 
                    : `<span style="color:var(--danger);"><i class="fas fa-times-circle"></i> İade Edildi</span>`;

                const returnButton = rental.durum === 'active' 
                    ? `<span class="return-action" onclick="returnBook(${rental.id})"><i class="fas fa-undo"></i> İade Al</span>`
                    : '<span style="color:#999;">Tamamlandı</span>';

                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td><strong>${rental.ad_soyad}</strong></td>
                    <td>${rental.kitap_adi || 'Kitap Bilgisi Yok'}</td>
                    <td>${new Date(rental.kiralama_tarihi).toLocaleDateString('tr-TR')}</td>
                    <td>${statusHtml}</td>
                    <td>${returnButton}</td>
                `;
                tbody.appendChild(tr);
            });

            if (dataToShow.length === 0) {
                tbody.innerHTML = '<tr><td colspan="5" style="text-align: center; color: #999;">Henüz kiralama kaydı bulunmuyor</td></tr>';
            }
        }

        async function renderBooks() {
            const tbody = document.getElementById('bookTableBody');
            tbody.innerHTML = "<tr><td colspan='6'>Yükleniyor...</td></tr>";
            
            const books = await getBooksFromAPI();

            tbody.innerHTML = '';
            
            books.forEach((book) => {
                const stockStatus = book.stok > 0 
                    ? `<span class="status-dot dot-green"></span> ${book.stok} Adet` 
                    : `<span class="status-dot dot-red"></span> Tükendi`;

                const coverImg = book.kapak_resmi && book.kapak_resmi.length > 5 
                    ? book.kapak_resmi 
                    : "https://via.placeholder.com/100x150?text=No+Cover";

                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td><img src="${coverImg}" class="book-thumb" alt="${book.baslik}"></td>
                    <td><strong>${book.baslik}</strong></td>
                    <td>${book.yazar}</td>
                    <td><span style="background:#eee; padding:4px 8px; border-radius:4px; font-size:0.8rem;">${book.kategori}</span></td>
                    <td>${stockStatus}</td>
                    <td>
                        <button class="action-btn btn-edit" onclick="openEditModal(${book.id}, '${book.baslik}', '${book.yazar}', '${book.kategori}', ${book.stok}, '${book.aciklama || ''}', '${book.ozet || ''}', '${book.kapak_resmi || ''}')">
                            <i class="fas fa-pen"></i> Düzenle
                        </button>
                        <button class="action-btn btn-delete" onclick="deleteBook(${book.id})">
                            <i class="fas fa-trash"></i> Sil
                        </button>
                    </td>
                `;
                tbody.appendChild(tr);
            });
        }

                const response = await fetch('api/admin_members.php');

                console.error('Üyeler yüklenirken hata:', error);

            }
        }


        async function addNewBook(e) {
            e.preventDefault();
            
            const bookData = {
                baslik: document.getElementById('addTitle').value,
                yazar: document.getElementById('addAuthor').value,
                kategori: document.getElementById('addCategory').value,
                aciklama: document.getElementById('addDescription').value,
                stok: parseInt(document.getElementById('addStock').value),
                kapak_resmi: document.getElementById('addCover').value
            };

            const result = await addBookToAPI(bookData);
            
            if (result.success) {
                closeModal('addBookModal');
                showToast("Yeni kitap başarıyla eklendi.");
                
                document.getElementById('addTitle').value = "";
                document.getElementById('addAuthor').value = "";
                document.getElementById('addDescription').value = "";
                document.getElementById('addSummary').value = "";
                document.getElementById('addCover').value = "";
                document.getElementById('addSummaryCounter').textContent = "0/500";
                
                await renderBooks();
                await updateStats();
            } else {
                showToast(result.message || "Kitap eklenirken hata oluştu.", "error");
            }
        }

        function openEditModal(id, title, author, category, stock, description, summary, cover) {
            document.getElementById('editId').value = id;
            document.getElementById('editTitle').value = title;
            document.getElementById('editAuthor').value = author;
            document.getElementById('editCategory').value = category;
            document.getElementById('editStock').value = stock;
            document.getElementById('editDescription').value = description || '';
            document.getElementById('editCover').value = cover || '';
            
            
            openModal('editBookModal');
        }

        async function saveEditBook(e) {
            e.preventDefault();
            
            const bookData = {
                id: document.getElementById('editId').value,
                baslik: document.getElementById('editTitle').value,
                yazar: document.getElementById('editAuthor').value,
                kategori: document.getElementById('editCategory').value,
                aciklama: document.getElementById('editDescription').value,
                stok: parseInt(document.getElementById('editStock').value),
                kapak_resmi: document.getElementById('editCover').value || ''
            };

            const result = await updateBookInAPI(bookData);
            
            if (result.success) {
                closeModal('editBookModal');
                showToast("Kitap bilgileri güncellendi.");
                await renderBooks();
                await updateStats();
            } else {
                showToast(result.message || "Kitap güncellenirken hata oluştu.", "error");
            }
        }

        async function deleteBook(bookId) {
            if(confirm("Bu kitabı envanterden silmek istiyor musunuz?")) {
                const result = await deleteBookFromAPI(bookId);
                
                if (result.success) {
                    showToast("Kitap silindi.");
                    await renderBooks();
                    await updateStats();
                } else {
                    showToast(result.message || "Kitap silinirken hata oluştu.", "error");
                }
            }
        }

        async function returnBook(rentalId) {
            if(confirm("Kitap fiziksel olarak teslim alındı mı?")) {
                const result = await returnBookAPI(rentalId);
                
                if (result.success) {
                    showToast("İade işlemi onaylandı. Stok güncellendi.");
                    await renderRentals();
                    await renderBooks();
                    await updateStats();
                } else {
                    showToast(result.message || "İade işlemi sırasında hata oluştu.", "error");
                }
            }
        }

        function switchTab(tabName, element) {
            if(element) {
                document.querySelectorAll('.menu-item').forEach(el => el.classList.remove('active'));
                element.classList.add('active');
            }
            
            if(tabName === 'dashboard') window.scrollTo({top:0, behavior: 'smooth'});
            if(tabName === 'books') document.getElementById('booksSection').scrollIntoView({behavior: 'smooth'});
            if(tabName === 'rentals') document.getElementById('rentalsSection').scrollIntoView({behavior: 'smooth'});
        }

        function openModal(id) {
            const el = document.getElementById(id);
            el.style.display = 'flex';
            setTimeout(() => el.classList.add('active'), 10);
        }

        function closeModal(id) {
            const el = document.getElementById(id);
            el.classList.remove('active');
            setTimeout(() => el.style.display = 'none', 300);
        }
        
        window.onclick = (e) => {
            if(e.target.classList.contains('modal-overlay')) closeModal(e.target.id);
        }

        function showToast(msg, type = "success") {
            const container = document.getElementById('toastContainer');
            const toast = document.createElement('div');
            toast.className = 'toast';
            toast.style.borderLeftColor = type === "success" ? 'var(--success)' : 'var(--danger)';
            toast.innerHTML = `
                <i class="fas fa-${type === "success" ? "check-circle" : "exclamation-circle"}" 
                   style="color:${type === "success" ? "var(--success)" : "var(--danger)"}; font-size:1.5rem;"></i>
                <div><h4>${type === "success" ? "Başarılı" : "Hata"}</h4><p>${msg}</p></div>`;
            container.appendChild(toast);
            setTimeout(() => toast.remove(), 3000);
        }

        function updateDate() {
            const dateElement = document.getElementById('current-date');
            const now = new Date();
            const options = { day: 'numeric', month: 'long', year: 'numeric' };
            dateElement.innerText = now.toLocaleDateString('tr-TR', options);
        }

        function updateCharCounter(textareaId, counterId) {
            const textarea = document.getElementById(textareaId);
            const counter = document.getElementById(counterId);
            
            if (textarea && counter) {
                const length = textarea.value.length;
                const maxLength = textarea.maxLength;
                
                counter.textContent = `${length}/${maxLength}`;
                
                counter.classList.remove('warning', 'danger');
                if (length > maxLength * 0.8 && length < maxLength) {
                    counter.classList.add('warning');
                } else if (length >= maxLength) {
                    counter.classList.add('danger');
                }
            }
        }
